def normalize_name(name):
    if not name:
        return ''
    n = name.lower().strip()
    for old, new in [('ā', 'a'), ('ä', 'a'), ('ī', 'i'), ('ü', 'u'), ('ū', 'u'), 
                      ('ṭ', 't'), ('ḍ', 'd'), ('ṇ', 'n'), ('ṅ', 'n'), ('ñ', 'n'), 
                      ('ṃ', 'm'), ('ṁ', 'm'), ('ḥ', 'h'), ('ś', 'sh'), ('ṣ', 'sh'), 
                      ('ṛ', 'ri'), ('ḷ', 'l')]:
        n = n.replace(old, new)
    return n

def names_match_list(expected, actual_list):
    if not expected or not actual_list:
        return False
    exp_lower = expected.lower().strip()
    exp_norm = normalize_name(expected)
    exp_target = NAME_VARIANTS.get(exp_lower, exp_lower)
    
    for actual in actual_list:
        act_lower = actual.lower().strip()
        act_norm = normalize_name(actual)
        if exp_lower == act_lower or exp_norm == act_norm:
            return True
        if actual == exp_target or act_lower == exp_target.lower():
            return True
        if normalize_name(exp_target) == act_norm:
            return True
        if len(exp_norm) >= 5 and len(act_norm) >= 5:
            if exp_norm in act_norm or act_norm in exp_norm:
                return True
    return False
