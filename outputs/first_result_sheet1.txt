# Load test data from CSV
def load_test_data(csv_path):
    verses = []
    with open(csv_path, 'r', encoding='utf-8') as f:
        content = f.read()
    
    lines = content.split('\n')
    current_verse = []
    
    for line in lines[2:]:
        line = line.strip()
        if not line:
            continue
        
        parts = line.rsplit(',', 1)
        if len(parts) == 2 and parts[1].strip() and not parts[1].strip().startswith('"'):
            chanda = parts[1].strip()
            if chanda and chanda not in ['', 'chanda']:
                verse_part = parts[0]
                if current_verse:
                    current_verse.append(verse_part)
                    full_verse = ' '.join(current_verse).replace('"', '').strip()
                    verses.append({'verse': full_verse, 'chanda': chanda})
                    current_verse = []
                else:
                    verse = verse_part.replace('"', '').strip()
                    if verse:
                        verses.append({'verse': verse, 'chanda': chanda})
        elif line.startswith('"') or (current_verse and not line.endswith(',')):
            current_verse.append(line.replace('"', ''))
    
    return verses

def normalize_chanda(name):
    name = name.lower().strip()
    replacements = {
        'ä': 'a', 'ā': 'a', 'ī': 'i', 'ū': 'u', 'ṛ': 'r', 'ṝ': 'r',
        'ṅ': 'n', 'ñ': 'n', 'ṭ': 't', 'ḍ': 'd', 'ṇ': 'n', 'ś': 's',
        'ṣ': 's', 'ḥ': 'h', 'ḷ': 'l', 'ṃ': 'm', 'ः': '', 'ी': 'i',
        'ा': 'a', 'ू': 'u', 'ं': 'm', 'ँ': '', 'ृ': 'r'
    }
    for old, new in replacements.items():
        name = name.replace(old, new)
    return name

def is_match(expected, identified):
    exp_norm = normalize_chanda(expected)
    id_norm = normalize_chanda(identified)
    
    if exp_norm in id_norm or id_norm in exp_norm:
        return True
    
    upajati_family = ['upajati', 'indravajra', 'upendravajra']
    if any(f in exp_norm for f in upajati_family):
        if any(f in id_norm for f in upajati_family):
            return True
    
    return False

def test_implementation(impl, verses, iterations=10):
    passed = 0
    
    # Accuracy test
    for v in verses:
        verse = v['verse']
        expected = v['chanda']
        
        try:
            verse_line = verse.split('।')[0].strip() if '।' in verse else verse.split('\n')[0].strip()
            result = impl.identify_line(verse_line)
            
            if result:
                if isinstance(result, list):
                    identified = result[0]['name'] if result else ''
                elif isinstance(result, dict):
                    identified = result.get('results', [{}])[0].get('name', '') if result.get('results') else ''
                else:
                    identified = ''
                
                if is_match(expected, identified):
                    passed += 1
        except:
            pass
    
    # Performance test
    start = time.perf_counter()
    for _ in range(iterations):
        for v in verses:
            verse = v['verse']
            try:
                verse_line = verse.split('।')[0].strip() if '।' in verse else verse.split('\n')[0].strip()
                impl.identify_line(verse_line)
            except:
                pass
    elapsed = time.perf_counter() - start
    avg_time = (elapsed / iterations / len(verses)) * 1000  # ms per verse
    
    return passed, avg_time
